cmake_minimum_required(VERSION 3.20)
project(sf3_runtime_checks CXX)
find_package(Threads REQUIRED)
add_executable(audio_buffer audio-buffer.cpp)
add_executable(game_speed game-speed.cpp)
target_include_directories(game_speed PRIVATE "${SATURN_RUNTIME}/src/host")
target_include_directories(audio_buffer PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../native")
set(CMAKE_CXX_STANDARD 20)
set(SATURN_NO_SDL ON CACHE BOOL "" FORCE)
if(NOT DEFINED SATURN_RUNTIME)
  message(FATAL_ERROR "Pass the framework runtime path")
endif()
add_subdirectory("${SATURN_RUNTIME}/tests" upstream)
add_executable(machine_regression machine-regression.cpp "${SATURN_RUNTIME}/tests/scu_test.cpp" "${SATURN_RUNTIME}/tests/main.cpp" "${SATURN_RUNTIME}/tests/no_modules.cpp")
target_include_directories(machine_regression PRIVATE "${SATURN_RUNTIME}/tests")
target_link_libraries(machine_regression saturn_core saturn_machine saturn_host Threads::Threads)
add_executable(slave_state_regression slave-state-regression.cpp "${SATURN_RUNTIME}/tests/main.cpp")
target_include_directories(slave_state_regression PRIVATE "${SATURN_RUNTIME}/tests")
target_link_libraries(slave_state_regression saturn_core saturn_machine saturn_host Threads::Threads)
add_executable(bup_regression "${SATURN_RUNTIME}/tests/bup_test.cpp" "${SATURN_RUNTIME}/tests/main.cpp" "${SATURN_RUNTIME}/tests/no_modules.cpp" bup-regression.cpp)
target_include_directories(bup_regression PRIVATE "${SATURN_RUNTIME}/tests")
target_link_libraries(bup_regression saturn_core saturn_machine saturn_host Threads::Threads)
add_executable(background_regression background-regression.cpp "${SATURN_RUNTIME}/tests/main.cpp" "${SATURN_RUNTIME}/tests/no_modules.cpp")
target_include_directories(background_regression PRIVATE "${SATURN_RUNTIME}/tests")
target_link_libraries(background_regression saturn_core saturn_machine saturn_host Threads::Threads)
add_executable(render_dump render-dump.cpp "${SATURN_RUNTIME}/tests/no_modules.cpp")
target_link_libraries(render_dump saturn_core saturn_machine saturn_host Threads::Threads)
target_include_directories(render_dump PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../native")
add_executable(hd_presentation hd-presentation.cpp)
target_include_directories(hd_presentation PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../native" "${SATURN_RUNTIME}/src/video")
add_executable(hd_assets hd-assets.cpp)
add_executable(hd_pack_index hd-pack-index.cpp)
target_include_directories(hd_pack_index PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../native" "${SATURN_RUNTIME}/src/video")
target_link_libraries(hd_pack_index PRIVATE Threads::Threads)
target_include_directories(hd_assets PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../native" "${SATURN_RUNTIME}/src/video")
target_link_libraries(hd_assets PRIVATE Threads::Threads)
target_link_libraries(hd_presentation PRIVATE Threads::Threads)
add_executable(core_regression core-regression.cpp "${SATURN_RUNTIME}/src/core/core.cpp")
add_executable(dsp_regression dsp-regression.cpp "${SATURN_RUNTIME}/src/core/core.cpp")
target_compile_definitions(dsp_regression PRIVATE SCUDSP_SOURCE="${SATURN_RUNTIME}/src/machine/scudsp.cpp")
target_include_directories(dsp_regression PRIVATE "${SATURN_RUNTIME}/include" "${SATURN_RUNTIME}/src/machine")
target_include_directories(core_regression PRIVATE "${SATURN_RUNTIME}/include")
if(WIN32)
  target_compile_options(runtime_tests PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/windows-test-env.h")
  target_compile_options(bup_regression PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/windows-test-env.h")
  if(NOT MINGW)
  get_filename_component(_clang_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
  file(GLOB _builtins "${_clang_dir}/../lib/clang/*/lib/windows/clang_rt.builtins-x86_64.lib")
  target_link_libraries(runtime_tests ${_builtins})
  target_link_libraries(bup_regression ${_builtins})
  target_link_libraries(machine_regression ${_builtins})
  target_link_libraries(background_regression ${_builtins})
  target_link_libraries(render_dump ${_builtins})
  endif()
endif()
add_executable(overlay_state_regression overlay-state-regression.cpp "${SATURN_RUNTIME}/tests/main.cpp")
target_include_directories(overlay_state_regression PRIVATE "${SATURN_RUNTIME}/tests" "${SATURN_RUNTIME}/src/machine")
target_link_libraries(overlay_state_regression saturn_core saturn_machine saturn_host Threads::Threads)
if(WIN32 AND NOT MINGW)
  target_link_libraries(overlay_state_regression ${_builtins})
endif()
